\begin{incode}
Optical signals are complex field envelopes around the optical carrier, scaled so that
$|E|^2$ is the instantaneous optical power in watts. Dual polarization uses the shape
$(B,2,n)$. Optical blocks maintain the following metadata:
\begin{itemize}
  \item \meta{wavelength_nm};
  \item \meta{power}, the average launch or received power in W;
  \item \meta{cd_ps_nm}, the accumulated chromatic dispersion $D\ell$;
  \item \meta{ase_psd}, the accumulated ASE density per polarization in W/Hz.
\end{itemize}
The OSNR is computed from the last two at any point of the link.
\end{incode}

\section{Transmitters}
\subsection{Directly modulated lasers}
An intensity modulator maps the electrical drive $x\in[-1,1]$ to the optical power
\begin{equation}
  P(t)=P_{\mathrm{avg}}\big(1+m\,x(t)\big),\qquad m=\frac{\mathrm{ER}-1}{\mathrm{ER}+1},
\end{equation}
since the extinction ratio is $\mathrm{ER}=P_1/P_0=(1+m)/(1-m)$. Three non-idealities are
modelled:
\begin{itemize}
  \item \textbf{Relative intensity noise}: additive Gaussian noise on $P$ with variance
  $\mathrm{RIN}\cdot P_{\mathrm{avg}}^2\cdot f_s/2$. As in the IEEE~802.3 link model \cite{cunningham2002}, RIN is
  referred to the average power.
  \item \textbf{Phase noise}: a Wiener process with the laser linewidth (\cref{eq:wiener}).
  \item \textbf{Transient chirp} of a directly modulated laser. The instantaneous frequency
  follows the power changes, $\Delta\nu(t)=\frac{\alpha}{4\pi}\frac{d}{dt}\ln P(t)$, with $\alpha$
  the Henry linewidth-enhancement factor. Integrating gives the phase
  \begin{equation}
    \phi_{\mathrm{chirp}}(t)=\frac{\alpha}{2}\ln\frac{P(t)}{P_{\mathrm{avg}}} .
  \end{equation}
\end{itemize}
The field is $E=\sqrt P\,e^{j\phi}$.

Two further effects of a directly modulated laser come from the rate equations
\cite{koch1984,coldren2012}.
\begin{itemize}
  \item \textbf{Adiabatic chirp.} The complete chirp is
  $\Delta\nu=\frac{\alpha}{4\pi}\big(\frac{d}{dt}\ln P+\kappa P\big)$. The second, adiabatic term
  is a frequency offset proportional to the power. It dominates the frequency excursion of DMLs
  and turns into intensity distortion after dispersive fiber. The block takes
  $\kappa_{\mathrm{ad}}=\alpha\kappa/(4\pi)$ in GHz/mW (\prm{adiabatic_chirp_ghz_mw}). It
  integrates $2\pi\kappa_{\mathrm{ad}}\!\int(P-P_{\mathrm{avg}})\,dt$ into the phase, so the carrier
  is defined at the average power. A test checks that a power step from $P_0$ to $P_1$ moves the
  instantaneous frequency by $\kappa_{\mathrm{ad}}(P_1-P_0)$.
  \item \textbf{Modulation bandwidth.} With \prm{resonance_ghz} set, the drive passes through the
  small-signal laser response of \cref{sec:mmf} (relaxation resonance, damping, parasitic pole),
  and its group delay is added to the \code{delay} metadata.
\end{itemize}
\codeexcerpt{im_mod}{Laser intensity modulation with bandwidth, RIN, phase noise and chirp (\blk{intensity_modulator}).}

\begin{pitfall}[Small-signal, not large-signal]
The laser dynamics are linearized: the bandwidth is the small-signal response around the bias,
and the chirp follows the Koch--Bowers relation. Large-signal effects of the full rate equations
(gain compression at high modulation depth, turn-on delay near threshold) and thermal effects
are not included.
\end{pitfall}

\subsection{Mach--Zehnder and IQ modulators}
A Mach--Zehnder modulator in push--pull configuration, biased at null, has the field transfer
function $E_{\mathrm{out}}/E_{\mathrm{in}}=\sin\big(\frac{\pi}{2}V/V_\pi\big)$. An IQ modulator
nests two of them, with a $90^\circ$ phase shift between the arms:
\begin{equation}
  \frac{E_{\mathrm{out}}}{E_{\mathrm{in}}}\propto\sin\Big(\frac\pi2\frac{V_I}{V_\pi}\Big)+j\sin\Big(\frac\pi2\frac{V_Q}{V_\pi}\Big).
\end{equation}
With the peak drive set to \prm{drive_swing}$\times V_\pi$, small swings are nearly linear,
while large swings compress the outer constellation points (\cref{fig:mzm}). The \prm{linear}
model maps the drive directly onto the field. The laser power is split equally between the
polarizations.

\section{Propagation in single-mode fiber}\label{sec:nlse}
\subsection{The nonlinear Schrödinger equation}
The slowly varying envelope $A(z,t)$ of a single-mode fiber, in a frame moving with the group
velocity, obeys
\begin{equation}
  \frac{\partial A}{\partial z}=
  \underbrace{-\frac{\alpha}{2}A-j\frac{\beta_2}{2}\frac{\partial^2A}{\partial t^2}}_{\hat D A}
  \;+\;\underbrace{j\gamma|A|^2A}_{\hat N A},
  \label{eq:nlse}
\end{equation}
with three ingredients~\cite{agrawal2019}:
\begin{itemize}
  \item the power attenuation $\alpha$, given in dB/km as \prm{attenuation_db_km};
  \item the group-velocity dispersion $\beta_2$, related to the dispersion parameter $D$
  (ps/nm/km) by
  \begin{equation}
    \beta_2=-\frac{D\lambda^2}{2\pi c}\qquad(\beta_2\approx\SI{-21.7}{ps^2/km}\ \text{for }D=17,\ \lambda=1550\,\mathrm{nm});
  \end{equation}
  \item the Kerr coefficient $\gamma$, given in 1/W/km as \prm{gamma_w_km}.
\end{itemize}
For wide WDM grids, or near the zero-dispersion wavelength, the third-order term
$\frac{\beta_3}{6}\partial_t^3A$ is added, with
$\beta_3=\big(\frac{\lambda}{2\pi c}\big)^2(\lambda^2S+2\lambda D)$ from the dispersion slope $S$
(\prm{dispersion_slope_ps_nm2_km}; about 0.057\,ps/nm$^2$/km for G.652 fiber at 1550\,nm~\cite{g652}).
It is 0 by default, giving the textbook NLSE.
With the Fourier convention of \cref{eq:fourier-convention}, $\partial_t\to j\omega$, so the
linear part has the exact solution
\begin{equation}
  \tilde A(z,\omega)=\tilde A(0,\omega)\exp\!\Big(j\frac{\beta_2}{2}\omega^2z-\frac{\alpha}{2}z\Big),
  \label{eq:dispersion-operator}
\end{equation}
an all-pass filter with quadratic phase. The dispersion therefore depends only on the
accumulated $D\ell$ (ps/nm), the quantity tracked in \meta{cd_ps_nm}.

\paragraph{Sign convention.} The envelope follows Agrawal's convention, in which the carrier is
$e^{-j\omega_0t}$. The dispersion, nonlinear and LO phases all use this convention consistently,
so solitons, SPM, CD compensation and frequency offsets behave correctly. With NumPy's transform,
however, a \emph{positive} baseband frequency is a \emph{lower} optical frequency (a longer
wavelength). This matters for odd-order terms. With slope, the operator is
$\exp\!\big(j(\tfrac{\beta_2}{2}\omega^2-\tfrac{\beta_3}{6}\omega^3)\ell\big)$ on the NumPy grid, which
gives the group delay $\tau=D\ell\,\Delta\lambda+\tfrac{S\ell}{2}\Delta\lambda^2$; the test suite
checks this to within 0.05\,ps. It also matters when WDM channel indices are mapped onto the
G.694.1 grid. The accumulated slope is tracked in \meta{cd_slope_ps_nm2}, and
\blk{cd_compensation} removes it together with $D\ell$.
\codeexcerpt{beta2}{From $D$ to $\beta_2$.}
\codeexcerpt{disp_op}{The dispersion operator of \cref{eq:dispersion-operator}.}

\paragraph{Dispersion alone.}
An unchirped Gaussian pulse $A(0,t)=e^{-t^2/(2T_0^2)}$ stays Gaussian and broadens as
\begin{equation}
  \frac{T_1(z)}{T_0}=\sqrt{1+\Big(\frac{z}{L_D}\Big)^2},\qquad L_D=\frac{T_0^2}{|\beta_2|}.
\end{equation}
For $T_0=5$\,ps and $|\beta_2|=\fiberbeta$\,ps$^2$/km, $L_D=\fiberLD$\,km.
\Cref{fig:dispersion} compares the \code{propagate} solver with this result. The simulated RMS
widths agree with the formula to all printed digits, as expected from an exact spectral method.

\paragraph{Kerr nonlinearity alone.}
Without dispersion \cref{eq:nlse} integrates exactly. The power profile is unchanged and the phase
grows in proportion to it, $A(L,t)=A(0,t)\,e^{-\alpha L/2}e^{j\gamma|A(0,t)|^2L_{\mathrm{eff}}}$ with
$L_{\mathrm{eff}}=(1-e^{-\alpha L})/\alpha$. This self-phase modulation (SPM) chirps the pulse and
broadens its spectrum, with a number of spectral peaks that grows with the peak phase
$\phi_{\max}=\gamma P_0L_{\mathrm{eff}}$ (\cref{fig:spm}).

\subsection{The split-step Fourier method}
With both terms present, \cref{eq:nlse} has no general closed form. Formally
$A(z+h)=e^{h(\hat D+\hat N)}A(z)$. The two operators do not commute, but each is trivial to
apply on its own: $\hat D$ is a multiplication in the frequency domain, and $\hat N$ a
multiplication in the time domain. The symmetric (Strang) splitting
\begin{equation}
  A(z+h)=e^{\frac h2\hat D}\,e^{h\hat N}\,e^{\frac h2\hat D}A(z)+\mathcal O(h^3)
  \label{eq:strang}
\end{equation}
is exact to second order, by the Baker--Campbell--Hausdorff formula: the first commutator
$[\hat D,\hat N]$ cancels by symmetry. Over a fixed length the $L/h$ local errors add up to a
global error $\mathcal O(h^2)$.

Attenuation is included in the half steps. The nonlinear phase is applied at mid-step to the
field whose power has been reduced by the first half step. The effective length that makes
this exact for a constant envelope is
\begin{equation}
  L_{\mathrm{eff}}(h)=\int_{-h/2}^{h/2}e^{-\alpha z'}dz'=\frac{2\sinh(\alpha h/2)}{\alpha} .
\end{equation}
For dual polarization the Manakov equation replaces $|A|^2$ by $|A_x|^2+|A_y|^2$ and $\gamma$ by
$\frac89\gamma$, the average over fast random polarization evolution.

\paragraph{Choosing the step.} A fixed step that suits 0\,dBm per channel, where the nonlinear
length $1/(\gamma P)$ is hundreds of km, is far too long for the 28\,dBm soliton of
\cref{fig:soliton}, whose nonlinear length is about 1\,km. A fixed 2\,km step leaves a 12\,\% error
there. The solver therefore uses the nonlinear-phase-rotation criterion of Sinkin et
al.~\cite{sinkin2003}. Each step is limited so that the peak nonlinear phase
$\gamma P_{\mathrm{peak}}L_{\mathrm{eff}}$ stays below \prm{nonlinear_phase_rad} (0.05\,rad by
default), and never exceeds $1/$\prm{steps_per_km}. The step lengthens automatically as the power
decays along the span. With the defaults the soliton error falls to $3\times10^{-4}$, while
typical WDM links keep 1--2\,km steps. Without \prm{nonlinear} the solver applies one exact FFT
for the whole span.

\subsection{Polarization-mode dispersion}
Random birefringence rotates the state of polarization and delays the two principal states by
the differential group delay (DGD). In a real fiber the birefringence changes along the length,
so the DGD is a random variable. It is Maxwellian distributed with mean
$\langle\tau\rangle=D_{\mathrm{PMD}}\sqrt\ell$, and it varies with frequency (higher-order PMD).
LinkStudio uses the coarse-step \emph{waveplate} model~\cite{marcuse1997}. A span is a cascade of
$N$ birefringent sections (\prm{pmd_sections}, 10 by default), each a fixed DGD $\delta\tau$
between Haar-random rotations:
\begin{equation}
  \mat T(\omega)=\mat U_{N+1}\prod_{k=N}^{1}\begin{pmatrix}e^{j\omega\delta\tau/2}&0\\0&e^{-j\omega\delta\tau/2}\end{pmatrix}\mat U_k,
  \qquad \delta\tau=\langle\tau\rangle\sqrt{\frac{3\pi}{8N}} .
\end{equation}
The total DGD then satisfies $\langle\tau^2\rangle=N\delta\tau^2$, and the Maxwellian relation
$\langle\tau^2\rangle=\frac{3\pi}{8}\langle\tau\rangle^2$ fixes $\delta\tau$. The rotations come from a QR
decomposition of a complex Gaussian matrix, with the phases of the diagonal of $\mat R$ removed.
The test suite extracts the DGD of 4000 fibers from the Jones matrix,
$\tau=\lambda_1-\lambda_2$ with $\lambda_i$ the eigenvalues of $j\,\mat T'\mat T^\dagger$. It finds the
mean within 1\,\% of $D_{\mathrm{PMD}}\sqrt\ell$ and $\langle\tau^2\rangle/\langle\tau\rangle^2=1.168$
against $3\pi/8=1.178$ (\cref{fig:pmd}).

\section{Optical amplification and OSNR}
\subsection{ASE noise}
An erbium-doped fiber amplifier of gain $G$ adds amplified spontaneous emission with power
spectral density, per polarization,
\begin{equation}
  S_{\mathrm{ASE}}=n_{\mathrm{sp}}\,h\nu\,(G-1),\qquad \mathrm{NF}=\frac{2n_{\mathrm{sp}}(G-1)}{G}\ \Rightarrow\ n_{\mathrm{sp}}=\frac{\mathrm{NF}\cdot G}{2(G-1)},
\end{equation}
where the noise figure NF is at least 2 (3\,dB) for $n_{\mathrm{sp}}\ge1$. In the simulation band
$f_s$, the ASE is circular Gaussian noise of variance $S_{\mathrm{ASE}}f_s$ per polarization.
\codeexcerpt{ase}{ASE density from gain and noise figure.}
\codeexcerpt{add_ase}{ASE noise in the simulated bandwidth.}

\subsection{Optical signal-to-noise ratio}
The OSNR is measured in a reference bandwidth $B_{\mathrm{ref}}=12.5$\,GHz (0.1\,nm at 1550\,nm),
counting the noise in both polarizations:
\begin{equation}
  \mathrm{OSNR}=\frac{P}{2S_{\mathrm{ASE}}B_{\mathrm{ref}}} .
\end{equation}
In a chain of $N$ identical spans, each followed by an amplifier compensating its loss
($G=e^{\alpha L_{\mathrm{span}}}$), the ASE adds up while the signal power is restored. With the
NF definition above, $2S_{\mathrm{ASE}}\approx\mathrm{NF}\,G\,h\nu$, so
\begin{equation}
  \mathrm{OSNR}_{\mathrm{dB}}\approx\underbrace{-10\log_{10}\frac{h\nu B_{\mathrm{ref}}}{1\,\mathrm{mW}}}_{58.0\ \mathrm{dB}}
  +P_{\mathrm{dBm}}-G_{\mathrm{dB}}-\mathrm{NF}_{\mathrm{dB}}-10\log_{10}N,
  \label{eq:osnr58}
\end{equation}
the ``58\,dB rule'' of link design.

\subsection{From OSNR to $E_s/N_0$}
For a coherent receiver whose noise is the ASE, the noise density per polarization is
$N_0=S_{\mathrm{ASE}}$ and the signal power per polarization is $P/p$ for $p$ polarizations.
Hence
\begin{equation}
  \frac{E_s}{N_0}=\frac{P/p}{S_{\mathrm{ASE}}R_s}=\mathrm{OSNR}\cdot\frac{2B_{\mathrm{ref}}}{p\,R_s},
\end{equation}
implemented as \code{osnr_to_esn0_db}. The test-suite uses it to check that the full coherent
DP-QPSK link (\cref{sec:coherent-dsp}) performs within about 0.3\,dB of theory.

\section{Wavelength-division multiplexing}
\blk{wdm_multiplexer} surrounds the channel under test with $N_{\mathrm{ch}}-1$ neighbours at
multiples of the channel spacing. The neighbours carry independent symbols with the same
constellation, pulse shape and power. Propagating the composite field through a nonlinear
fiber produces cross-phase modulation (the power of the neighbours modulates the phase of the
channel under test) and four-wave mixing. \blk{wdm_demultiplexer} selects a channel with a
second-order super-Gaussian optical filter, $|H(f)|=\exp\!\big(-\tfrac{\ln2}{2}(2f/B)^4\big)$, which
is 3\,dB down at $\pm B/2$. The sample rate must cover all channels. The \emph{WDM with fiber
nonlinearity} scenario shows the MER of the centre channel dropping by several dB when the

\section{Short-reach links: VCSELs and multimode fiber}\label{sec:mmf}
Inside a data center most links are shorter than a few hundred metres. They use 850\,nm
vertical-cavity surface-emitting lasers (VCSELs) over graded-index multimode fiber (MMF). The
50\,\textmu m core makes coupling and connectors cheap, but it guides hundreds of modes.
100GBASE-SR4, for example, sends four lanes of 25.78\,GBd NRZ over 70\,m of OM3 or 100\,m of
OM4. The reach is set by dispersion, not loss. This section builds the fiber model from the
mode theory of graded-index cores \cite{gloge1973,olshansky1976}, links it to the bandwidth
model of the IEEE~802.3 link model \cite{cunningham2002}, and adds the transmitter and receiver
of such a link. The blocks are \blk{vcsel} and \blk{multimode_fiber}, in
\file{backend/linkstudio/phy/multimode.py}.

\subsection{Why the fiber acts on power}
Write the field in the fiber as a sum of guided modes, $E(\mathbf r,t)=\sum_m a_m(t)\,\psi_m(\mathbf r)$,
with orthonormal mode profiles, $\int\psi_m\psi_n^*\,dA=\delta_{mn}$. A photodiode that collects
the whole core measures
\begin{equation}
  P_{\mathrm{out}}(t)=\int|E|^2\,dA=\sum_m|a_m(t)|^2 :
\end{equation}
the cross terms between modes vanish by orthogonality. Each mode is excited with a fraction
$w_m$ of the launched power and arrives after its own group delay $\tau_m$, so
\begin{equation}
  P_{\mathrm{out}}(t)=10^{-\alpha_{\mathrm{dB}}L/10}\sum_m w_m\,P_{\mathrm{in}}(t-\tau_m)
  =10^{-\alpha_{\mathrm{dB}}L/10}\,(h*P_{\mathrm{in}})(t).
  \label{eq:mmf-power}
\end{equation}
The fiber is a linear system for the optical \emph{power}, not for the field. Chromatic dispersion
fits the same picture. A multimode VCSEL has a spectrum about 0.4\,nm wide (RMS, roughly
170\,GHz), far wider than the modulation, so its spectral components add incoherently. A
component at wavelength $\lambda$ is delayed by $D L(\lambda-\lambda_c)$, so the intensity response is
the source spectrum mapped onto delays. For a Gaussian spectrum of RMS width $\sigma_\lambda$ this is
a Gaussian of RMS width $|D|L\sigma_\lambda$. The total response is the modal response convolved with
the chromatic one, $H(f)=H_{\mathrm{modal}}(f)\,H_{\mathrm{cd}}(f)$, and the variances add.

\begin{keypoint}
SMF links (\cref{sec:nlse}) propagate a coherent field and dispersion is an all-pass filter on
the field. VCSEL--MMF links propagate incoherent power and dispersion is a low-pass filter on
the intensity. \blk{multimode_fiber} therefore filters $|E|^2$ and rebuilds the field as
$\sqrt{P_{\mathrm{out}}}$ with the input phase. It is meant for direct detection.
\end{keypoint}

\subsection{Mode groups of an $\alpha$-profile core}
Graded-index cores follow a power law,
$n^2(r)=n_1^2\big[1-2\Delta(r/a)^\alpha\big]$ for $r<a$, with core radius $a$, peak index $n_1$ and
contrast $\Delta=\mathrm{NA}^2/(2n_1^2)$. For the standard 50\,\textmu m, NA\,=\,0.2 fiber,
$\Delta=\mmfdelta$. In the WKB approximation \cite{gloge1973} the number of modes whose
propagation constant exceeds $\beta$ is
\begin{equation}
  m(\beta)=M\Big[\frac{n_1^2k^2-\beta^2}{2\Delta n_1^2k^2}\Big]^{\frac{\alpha+2}{\alpha}},\qquad
  M=\frac{\alpha}{\alpha+2}\,a^2k^2n_1^2\Delta=\frac{\alpha}{\alpha+2}\,\frac{V^2}{2},
\end{equation}
with $V=ka\,\mathrm{NA}$. Inverting it gives the propagation constant of the mode with label
$m$: $\beta=n_1k\sqrt{1-2\Delta x}$, where $x=(m/M)^{\alpha/(\alpha+2)}$.

\paragraph{Group delay.} The delay per unit length is $\tau/L=c^{-1}\,d\beta/dk$. Keep $m$ fixed
(the same mode) and note that $M\propto k^2n_1^2$. Then
$dx/dk=-\frac{\alpha}{\alpha+2}\,x\,\frac{2N_1}{n_1k}$, with group index $N_1=d(n_1k)/dk$, and
\begin{equation}
  \frac{d\beta}{dk}=N_1\sqrt{1-2\Delta x}+\frac{n_1k\,\Delta}{\sqrt{1-2\Delta x}}\,\frac{\alpha}{\alpha+2}\,\frac{2N_1x}{n_1k}
  \;\Longrightarrow\;
  \tau=\frac{LN_1}{c}\;\frac{1-\frac{4\Delta x}{\alpha+2}}{\sqrt{1-2\Delta x}} .
  \label{eq:mmf-exact}
\end{equation}
Since $\Delta\approx0.01$, expanding to second order loses nothing. Including the profile
dispersion $\epsilon=-\frac{2n_1}{N_1}\frac{\lambda}{\Delta}\frac{d\Delta}{d\lambda}$ gives the
Olshansky--Keck form \cite{olshansky1976}:
\begin{equation}
  \frac{\tau}{LN_1/c}-1=\Delta\,C_1\,x+\Delta^2C_2\,x^2,\qquad
  C_1=\frac{\alpha-2-\epsilon}{\alpha+2},\quad C_2=\frac{3\alpha-2-2\epsilon}{2(\alpha+2)} .
  \label{eq:mmf-delay}
\end{equation}
With $\epsilon=0$, \cref{eq:mmf-delay} is the Taylor expansion of \cref{eq:mmf-exact}. The test
suite checks that the two agree to $\mathcal O(\Delta^3)$.

\paragraph{Overfilled RMS width and the optimum profile.} If every mode carries the same power,
$u=m/M$ is uniform on $[0,1]$. Then $x=u^q$ with $q=\alpha/(\alpha+2)$ has moments
$\mathbb E[x^k]=1/(kq+1)$, and the RMS spread of \cref{eq:mmf-delay} follows in closed form:
\begin{equation}
  \frac{\sigma_\tau}{LN_1/c}=\Delta\Big[C_1^2\operatorname{Var}x+2\Delta C_1C_2\operatorname{Cov}(x,x^2)+\Delta^2C_2^2\operatorname{Var}x^2\Big]^{1/2}.
\end{equation}
The leading term vanishes for $\alpha=2+\epsilon$. Minimizing the whole expression gives
\cite{olshansky1976}
\begin{equation}
  \alpha_{\mathrm{opt}}=2+\epsilon-\Delta\frac{(4+\epsilon)(3+\epsilon)}{5+2\epsilon},\qquad
  \sigma_{\min}\approx\frac{LN_1\Delta^2}{20\sqrt3\,c},
\end{equation}
that is $\alpha_{\mathrm{opt}}=\mmfalphaopt$ and $\sigma_{\min}=\mmfsigmamin$\,ps/km for our fiber.
A parabolic core ($\alpha=2$) is already five times worse. A profile error of only $\pm0.01$ in
$\alpha$ multiplies the spread by two to three (\cref{fig:mmf-profile}). This sensitivity is why
real fibers are graded by measured bandwidth rather than by profile.

\paragraph{Discrete groups.} Modes with equal principal mode number are degenerate and travel
together. A core holds about $G=\sqrt M$ such groups: \mmfgroups{} for OM4 at 850\,nm. Group
$\mu$ ($\mu=1\dots G$) holds the $2\mu-1$ modes between $u=((\mu-1)/G)^2$ and $(\mu/G)^2$. The
block evaluates \cref{eq:mmf-delay} at the group centre and removes the mean delay, so the
response does not include the bulk latency.

\codeexcerpt{mmf_groups}{Group delays and launch weights.}

\subsection{Launch conditions}
The power distribution $w_\mu$ depends on how light enters the fiber.
\begin{itemize}
  \item \textbf{Overfilled launch} (every mode equally): $w_\mu\propto2\mu-1$. It excites the
  outer groups strongly. The \emph{overfilled launch} (OFL) bandwidth is defined with it.
  \item \textbf{Gaussian beam} (\prm{launch}\code{=gaussian}). In a parabolic core the modes are
  the Hermite--Gauss modes of a two-dimensional harmonic oscillator, with fundamental field
  radius $w_0=\sqrt{a\lambda/(\pi\,\mathrm{NA})}\approx5.8$\,\textmu m. Group $p=\mu-1$ collects
  the modes with $n_x+n_y=p$. A beam of radius $w$ displaced by $r_0$ along $x$ separates into
  $x$ and $y$ factors, so the group powers follow from one-dimensional overlap integrals,
  computed numerically in the block.
\end{itemize}
Two cases have exact closed forms and serve as checks (\cref{fig:mmf-launch}):
\begin{itemize}
  \item a matched beam ($w=w_0$) displaced by $r_0$ is a coherent state of the oscillator, so the
  group powers are Poisson distributed with mean $r_0^2/w_0^2$;
  \item a centred beam of the wrong size is a squeezed state, which excites only even groups,
  with geometric weights $(1-t)\,t^{p/2}$, $t=\big((w^2-w_0^2)/(w^2+w_0^2)\big)^2$.
\end{itemize}

the \emph{effective modal bandwidth} (EMB) used for VCSEL links exceeds the OFL bandwidth
(\cref{tab:om}).

\paragraph{Gaussian bandwidth model} (\prm{model}\code{=bandwidth}, the default). The IEEE~802.3
link model \cite{cunningham2002} assumes Gaussian responses. For
$H(f)=\exp(-2\pi^2\sigma^2f^2)$ the 3-dB optical bandwidth satisfies $H(B)=\frac12$, hence
\begin{equation}
  B\,\sigma=\frac{\sqrt{\ln2/2}}{\pi}\approx0.187 .
\end{equation}
The modal width over a length $L$ is $\sigma_{\mathrm{modal}}=0.187\,L/\mathrm{EMB}$ (linear length
scaling, as in the link model). The chromatic width uses the dispersion of the fiber at the
source wavelength, $D_1=\frac{S_0\lambda}{4}\big(1-(\lambda_0/\lambda)^4\big)$, about
$-100$\,ps/(nm\,km) at 850\,nm for $\lambda_0=1320$\,nm and $S_0=0.101$\,ps/(nm$^2$\,km). It adds a
second-order term $D_2=0.7S_0\sigma_\lambda$ for the spectral width:
\begin{equation}
  \sigma_{\mathrm{cd}}=L\,\sigma_\lambda\sqrt{D_1^2+D_2^2}.
\end{equation}
\codeexcerpt{mmf_chromatic}{Chromatic RMS width of the IEEE~802.3 link model.}

\paragraph{Mode-group model} (\prm{model}\code{=mode_groups}). The response is the sum of
delayed impulses of \cref{eq:mmf-power}, multiplied by the Gaussian chromatic response.
Without an explicit \prm{alpha}, the block detunes the profile from $\alpha_{\mathrm{opt}}$ until
the overfilled RMS width of the discrete groups equals $0.187/\mathrm{OFL}$. The result is
$\alpha=\mmfalphaomfour$ for OM4 and $\alpha=\mmfalphaomthree$ for OM3. The fiber then meets its
class specification by construction, and the launch options show how far a real transmitter
improves on the overfilled launch.
\codeexcerpt{mmf_response}{Power transfer function of the fiber.}

The spectral width comes from the \code{spectral_width_nm} metadata set by \blk{vcsel}, or from
\prm{source_rms_width_nm}. Filtering uses \code{apply_frequency_response} on $|E|^2$. Dual
polarization inputs are summed into one intensity. The metrics report the modal and chromatic
RMS widths, the resulting Gaussian-equivalent bandwidth and, for a Gaussian launch, the fraction
of power coupled into guided modes (also applied as a loss).
\end{incode}

\subsection{VCSEL transmitter and reference receiver}
Linearizing the laser rate equations around the bias point gives the small-signal modulation
response of a semiconductor laser, a damped second-order resonance. The RC parasitics of the
chip add a pole \cite{coldren2012}:
\begin{equation}
  H(f)=\frac{f_r^2}{f_r^2-f^2+jf\gamma/(2\pi)}\cdot\frac{1}{1+jf/f_p},
\end{equation}
with relaxation-oscillation frequency $f_r$, damping rate $\gamma$ and parasitic frequency $f_p$.
Without parasitics and with weak damping the 3-dB bandwidth tends to $\sqrt{1+\sqrt2}\,f_r\approx1.55f_r$,
a limit the test suite checks. The default 25G-class device ($f_r=20$\,GHz,
$\gamma=100$\,ns$^{-1}$, $f_p=20$\,GHz) has a bandwidth of 22.6\,GHz. \blk{vcsel} normalizes and
filters the drive, maps it to power with the extinction ratio (4\,dB by default, typical of
VCSELs), clips negative power (the laser stops emitting below threshold) and adds RIN. The group
delay of $H$ at DC is added to the \code{delay} metadata, so the receiver samples at the eye
centre.
\codeexcerpt{vcsel_response}{Small-signal VCSEL response.}

Optical Ethernet specifies receivers through a reference response: a fourth-order
Bessel--Thomson low-pass at $0.75R_s$. \blk{photodiode} applies it when \prm{bandwidth_ghz} is set.
The filter acts on signal and noise. The reported \code{noise_var} is scaled by the noise
bandwidth of the filter, $\frac1{f_s}\int|H|^2df$, so the result no longer depends on the
simulation sample rate.

\begin{pitfall}[What the multimode model leaves out]
The IEEE link model also budgets mode-partition noise (the VCSEL modes fluctuate in power while
the total stays constant, which dispersion turns into noise) and modal noise (speckle filtered
by connector offsets). Neither is modelled yet. The mode-group model also ignores the coupling
between modal and chromatic dispersion (each group has its own wavelength dependence
\cite{golowich2003}), differential mode attenuation and mode coupling. These effects are small for
the short, well-made links the classes describe. Statistically realistic fibers would need DMD
templates measured according to IEC~60793-1-49.
\end{pitfall}

\section{Free-space optics}
\subsection{Scintillation}
Atmospheric turbulence makes the received intensity fluctuate (scintillation). \blk{fso_turbulence}
multiplies the field by $\sqrt h$, where the normalized irradiance $h$ ($\E h=1$) is constant
over a frame and follows one of two models.
\begin{itemize}
  \item \textbf{Log-normal} (weak turbulence): $\ln h\sim\N(-\sigma^2/2,\sigma^2)$.
  \item \textbf{Gamma-gamma} (moderate to strong turbulence): $h$ is the product of two
  independent unit-mean gamma variables with parameters $\alpha$ (large eddies) and $\beta$
  (small eddies). Its density is
  \begin{equation}
    f(h)=\frac{2(\alpha\beta)^{\frac{\alpha+\beta}{2}}}{\Gamma(\alpha)\Gamma(\beta)}\,h^{\frac{\alpha+\beta}{2}-1}K_{\alpha-\beta}\big(2\sqrt{\alpha\beta h}\big).
  \end{equation}
\end{itemize}

The parameters follow from the turbulence strength. For a plane wave the Rytov variance is
$\sigma_R^2=1.23\,C_n^2k^{7/6}L^{11/6}$, with $C_n^2$ the refractive-index structure parameter
(about $10^{-16}$ to $10^{-13}$\,m$^{-2/3}$ at optical wavelengths~\cite{p1814}). Al-Habash, Andrews and
Phillips~\cite{alhabash2001} give
\begin{equation}
  \alpha=\Big[e^{0.49\sigma_R^2/(1+1.11\sigma_R^{12/5})^{7/6}}-1\Big]^{-1},\qquad
  \beta=\Big[e^{0.51\sigma_R^2/(1+0.69\sigma_R^{12/5})^{5/6}}-1\Big]^{-1}.
\end{equation}
For example, $\sigma_R^2=1$ gives $\alpha=4.39$ and $\beta=2.56$. The scintillation index of the
gamma-gamma model is $\sigma_I^2=1/\alpha+1/\beta+1/(\alpha\beta)$. With \prm{rytov_variance} set,
\blk{fso_turbulence} derives $\alpha$ and $\beta$, and the log-normal variance
$\ln(1+\sigma_I^2)$, so both models share the same scintillation index. In weak turbulence that
index tends to $\sigma_R^2$.

\subsection{The complete link: \blk{fso_channel}}
\blk{fso_channel} computes the whole budget of ITU-R P.1814~\cite{p1814} from geometry and weather.
\begin{itemize}
  \item \textbf{Atmospheric attenuation} from the visibility $V$ (2\,\% contrast threshold, km),
  with Kim's size-distribution exponent:
  \begin{equation}
    \gamma_{\mathrm{sp}}=\frac{17}{V}\Big(\frac{0.55\,\mu\mathrm m}{\lambda}\Big)^{q}\ \mathrm{dB/km},\qquad
    q=\begin{cases}1.6 & V>50\\ 1.3 & 6<V\le50\\ 0.16V+0.34 & 1\le V\le6\\ V-0.5 & 0.5\le V<1\\ 0 & V<0.5 .\end{cases}
  \end{equation}
  Light fog ($V=1$\,km) costs 10\,dB/km at 1550\,nm; haze at $V=5$\,km about 1\,dB/km.
  \item \textbf{Geometric loss}: the beam of full divergence $\theta$ covers the area
  $\frac\pi4(d\theta)^2$, so a receiver of diameter $D_R$ collects
  $A_{\mathrm{geo}}=20\log_{10}(d\theta/D_R)$\,dB, or 0 when the aperture catches the whole beam.
  \item \textbf{Scintillation} from $C_n^2$ via $\sigma_R^2$, as above.
  \item \textbf{Pointing errors}~\cite{farid2007}. Building sway gives a radial beam offset
  $r\sim\mathrm{Rayleigh}(\sigma_s)$ on the receiver plane. A Gaussian beam of radius
  $w_z=d\theta/2$ on an aperture of radius $a$ then collects
  $h_p=e^{-2r^2/w_{z,\mathrm{eq}}^2}$ relative to perfect alignment, with
  $w_{z,\mathrm{eq}}^2=w_z^2\sqrt\pi\,\mathrm{erf}(v)/(2v\,e^{-v^2})$ and $v=\sqrt{\pi/2}\,a/w_z$. On
  average $\E h_p=w_{z,\mathrm{eq}}^2/(w_{z,\mathrm{eq}}^2+4\sigma_s^2)$, which the test suite checks.
\end{itemize}
All fading terms are quasi-static (one value per frame) and appear on \code{csi}. The metrics give
each loss term and $\sigma_R^2$. Aperture averaging, beam wander and time-correlated fading are not
modelled. The \emph{Free-space optical link budget} scenario runs a 1\,Gb/s OOK link over 1\,km
with all four effects.

\section{Direct detection}
\subsection{PIN and APD receivers}
A photodiode converts power into current, $I=\mathcal RP$, with responsivity $\mathcal R$
(A/W). Over the noise bandwidth $B=f_s/2$ (or through the reference receiver filter of
\cref{sec:mmf} when \prm{bandwidth_ghz} is set) the current carries three contributions:
\begin{itemize}
  \item \textbf{shot noise}, $\sigma_{\mathrm{sh}}^2=2qM^2F\,(\mathcal RP+I_d)B$;
  \item \textbf{thermal noise}, $\sigma_{\mathrm{th}}^2=i_n^2B$, with $i_n$ the input-referred
  current density (\prm{thermal_noise_pa_rthz});
  \item \textbf{avalanche gain} (APD): the multiplication $M$ comes with McIntyre's excess noise
  factor $F=kM+(1-k)(2-1/M)$, with $k$ the ionization ratio.
\end{itemize}
The APD gain raises the signal above thermal noise but amplifies shot noise by $M^2F$, so there
is an optimum $M$.
\codeexcerpt{photodiode}{Square-law detection with shot, thermal and APD excess noise.}

\subsection{OOK sensitivity and the Q factor}
For on--off keying with levels $I_{1,0}=\mathcal RP_{1,0}$ and Gaussian noise of standard
deviations $\sigma_{1,0}$, the optimum threshold gives
\begin{equation}
  \mathrm{BER}=\frac12\erfc\Big(\frac{Q}{\sqrt2}\Big),\qquad Q=\frac{I_1-I_0}{\sigma_1+\sigma_0},
  \qquad P_{1,0}=\frac{2P_{\mathrm{avg}}}{1+\mathrm{ER}^{\mp1}} .
\end{equation}
Thermal noise is signal independent, so $\sigma_1=\sigma_0$ and $Q\propto P_{\mathrm{avg}}$: the BER
falls by many decades within a few dB (\cref{fig:ook}). The receiver sensitivity is
conventionally quoted at $10^{-9}$ ($Q=6$) or at a FEC threshold.

\section{Coherent detection}
\subsection{The coherent front end}
A coherent receiver mixes the signal with a local oscillator (LO) in a $90^\circ$ hybrid and
detects the four outputs with two balanced photodiode pairs per polarization
(\cref{fig:coherent-fe}). Balanced detection cancels the direct-detection terms $|E|^2$ and
$|E_{\mathrm{LO}}|^2$ and leaves the beat term:
\begin{equation}
  i_I+j\,i_Q=\mathcal R\sqrt{P_{\mathrm{LO}}}\;E(t)\,e^{-j(2\pi\Delta f\,t+\phi_{\mathrm{LO}}(t))}+n(t).
\end{equation}
The field is thus translated linearly to the electrical domain, including its phase and
polarization. With dual polarization a splitter feeds each hybrid with half of the LO, so
$P_{\mathrm{LO}}$ above is \prm{lo_power_dbm} divided by the number of polarizations
(\cref{fig:coherent-fe}). With a strong LO, the LO shot noise $2q\mathcal RP_{\mathrm{LO}}B$ dominates
the thermal noise. The signal and the shot noise then scale together, so the LO split does not
change the shot-noise-limited SNR, but it does change the margin over thermal noise.

The simulator processes each frame with circular FFTs (fiber, dispersion compensation). An LO

block therefore makes the LO phase periodic:
\begin{itemize}
  \item the frequency offset is rounded to the frame's frequency grid $f_s/n$;
  \item the Wiener phase is turned into a Brownian bridge, $\phi_k-\frac kn\phi_n$, which has the
  same local increments but returns to zero.
\end{itemize}
\end{incode}

\paragraph{Chromatic dispersion compensation.}
The accumulated dispersion is known from the metadata (or given), and its all-pass response
\cref{eq:dispersion-operator} is inverted exactly by multiplying with its conjugate in the
frequency domain.
\codeexcerpt{cd_comp}{Frequency-domain CD compensation.}

\paragraph{Frequency-offset estimation.}
Raising QPSK to the fourth power removes the modulation: $(s\,e^{j2\pi\Delta f t})^4=s^4e^{j2\pi4\Delta ft}$
with $s^4=-1$ for every QPSK symbol, and approximately constant on average for square QAM. The
spectrum of $r^4$ therefore has a line at $4\Delta f$. The estimate is the position of its peak
divided by 4, refined by zero-padding and computed on a low-pass filtered copy to reject noise.
In dual polarization a rotated mixture $as_1+bs_2$ has a fourth-power line proportional to
$a^4+b^4$, which can vanish. The estimator therefore also uses the combinations
$(x_0+x_1)/\sqrt2$ and $(x_0+jx_1)/\sqrt2$.

Compensating the offset \emph{after} the dispersion compensator leaves a residual: the CD
filter was applied to a spectrum shifted by $\Delta\omega$, so the error is
$\exp\!\big(j\frac{\beta_2L}{2}(2\omega\Delta\omega+\Delta\omega^2)\big)$, a bulk delay
$\beta_2L\Delta\omega$ plus a constant phase. The block removes it.
\codeexcerpt{fo_est}{Fourth-power FO estimation over several polarization combinations.}

\paragraph{Adaptive $2\times2$ equalization.}
Polarization rotation, PMD and residual dispersion form a $2\times2$ MIMO channel, inverted by
a fractionally spaced butterfly of four FIR filters (\cref{fig:butterfly}):
\begin{equation}
  y_p[k]=\sum_{q\in\{x,y\}}\vect w_{pq}\T\,\vect u_q[k],\qquad
  \vect w_{pq}\leftarrow\vect w_{pq}+\mu\,e_p[k]\,\vect u_q\conj[k] .
\end{equation}
The error $e_p$ is either:
\begin{itemize}
  \item the constant-modulus error $y_p(R-|y_p|^2)$ with $R=\E|s|^4/\E|s|^2$ (Godard's
  CMA~\cite{godard1980}), which is blind and insensitive to the carrier phase; or
  \item the LMS error $d_p-y_p$, trained on known symbols and then decision-directed, with a
  larger step during training.
\end{itemize}

\paragraph{Carrier phase recovery.}
Laser phase noise of $2\times100$\,kHz at 32\,GBd is slow, but large enough to need
feed-forward tracking. Two estimators are available.
\begin{itemize}
  \item The Viterbi--Viterbi estimator removes the QPSK modulation with the fourth power:
  $\hat\phi=\frac14\big(\arg\sum_{|i|\le W}y_{k+i}^4-\pi\big)$.
  \item Blind phase search (BPS)~\cite{pfau2009} works for any square QAM. It rotates the
  symbols by $B$ test phases in $[-\pi/4,\pi/4)$, sums over a window the squared distances to
  the nearest constellation points, and keeps the phase with the smallest sum.
\end{itemize}
Both estimates are defined modulo $\pi/2$; they are unwrapped (on $4\hat\phi$) to follow the phase
continuously. The remaining quadrant ambiguity is resolved once per frame with a short block of
known symbols, taken after the equalizer has converged.
\codeexcerpt{bps}{Viterbi--Viterbi and blind phase search.}
